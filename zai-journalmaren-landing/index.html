[ 02 / snippet ] A pattern I keep reaching for

A small Go utility I copy into almost every service. Rate-limits per-key work without external dependencies.

go ratelimit/tokenbucket.go 
// TokenBucket is a per-key rate limiter safe for concurrent use.
// Capacity=10, refill=1/sec → ~1 QPS sustained, burst of 10.
type TokenBucket struct {
    mu       sync.Mutex
    capacity float64
    refill   float64 // tokens per second
    buckets  map[string]*bucket
}

func (tb *TokenBucket) Allow(key string) bool {
    tb.mu.Lock()
    defer tb.mu.Unlock()

    b, ok := tb.buckets[key]
    if !ok {
        b = &bucket{tokens: tb.capacity, last: time.Now()}
        tb.buckets[key] = b
    }
    return b.take(tb.capacity, tb.refill)
}
